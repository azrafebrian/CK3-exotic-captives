# hidden event to delay arrival
halfsade_captives.0001 = {
	type = character_event
	hidden = yes

	immediate = {
		send_interface_toast = {
			title = halfsade_captives.0001.toast
			custom_tooltip = halfsade_captives.0001.toast_desc
		}
		trigger_event = {
			id = halfsade_captives.0002
			days = 5
		}
	}
}

# captive selector
halfsade_captives.0003 = {
	type = character_event
	title = halfsade_captives.0003.t
	desc = halfsade_captives.0003.desc
	theme = prison

	left_portrait = {
		character = scope:halfsade_captive_1
		animation = prisonhouse
	}
	lower_center_portrait = {
		character = scope:halfsade_captive_2
		animation = prisonhouse
	}
	right_portrait = {
		character = scope:halfsade_captive_3
		animation = prisonhouse
	}

	immediate = {
		save_scope_as = halfsade_home_ruler

		if = {
			limit = { NOT = { has_variable = halfsade_captives_left_1 } }

			halfsade_captives_pick_source_effect = { SLOT = 1 }
			halfsade_captives_pick_source_effect = { SLOT = 2 }
			halfsade_captives_pick_source_effect = { SLOT = 3 }
			halfsade_captives_clear_choice_flags_effect = yes

			halfsade_captives_create_captive_effect = { SLOT = 1 }
			halfsade_captives_create_captive_effect = { SLOT = 2 }
			halfsade_captives_create_captive_effect = { SLOT = 3 }

			set_variable = { name = halfsade_captives_left_1 value = scope:halfsade_captive_1 }
			set_variable = { name = halfsade_captives_left_2 value = scope:halfsade_captive_2 }
			set_variable = { name = halfsade_captives_left_3 value = scope:halfsade_captive_3 }
		}
		else = {
			var:halfsade_captives_left_1 = { save_scope_as = halfsade_captive_1 }
			var:halfsade_captives_left_2 = { save_scope_as = halfsade_captive_2 }
			var:halfsade_captives_left_3 = { save_scope_as = halfsade_captive_3 }
		}
	}

	option = {
		name = halfsade_captives.0003.a
		trigger = { short_term_gold >= halfsade_captives_price_3 }
		show_as_unavailable = { always = yes }
		halfsade_captives_pay_effect = { COUNT = 3 }
		halfsade_captives_welcome_effect = { SLOT = 1 }
		halfsade_captives_welcome_effect = { SLOT = 2 }
		halfsade_captives_welcome_effect = { SLOT = 3 }
		halfsade_captives_cleanup_effect = yes
	}

	option = {
		name = halfsade_captives.0003.b
		trigger_event = halfsade_captives.0008
	}

	option = {
		name = halfsade_captives.0003.c
		trigger = { short_term_gold >= halfsade_captives_price_3 }
		show_as_unavailable = { always = yes }
		halfsade_captives_pay_effect = { COUNT = 3 }
		custom_tooltip = halfsade_captives.release_tt
		halfsade_captives_release_effect = { SLOT = 1 }
		halfsade_captives_release_effect = { SLOT = 2 }
		halfsade_captives_release_effect = { SLOT = 3 }
		halfsade_captives_cleanup_effect = yes
	}

	option = {
		name = halfsade_captives.0003.d
		halfsade_captives_send_home_effect = { SLOT = 1 }
		halfsade_captives_send_home_effect = { SLOT = 2 }
		halfsade_captives_send_home_effect = { SLOT = 3 }
		halfsade_captives_cleanup_effect = yes
	}
}

halfsade_captives.0006 = {
	type = character_event
	hidden = yes

	trigger = {
		has_character_modifier = halfsade_captives_freed_modifier
		is_ruler = no
		exists = court_owner
	}

	immediate = {
		court_owner = { save_scope_as = halfsade_host }
		if = {
			limit = { halfsade_captives_stays_trigger = yes }
			remove_character_modifier = halfsade_captives_freed_modifier
		}
		else = {
			add_character_flag = halfsade_captives_leaving_pending
			scope:halfsade_host = { trigger_event = halfsade_captives.0007 }
		}
	}
}

# captives leaving
halfsade_captives.0007 = {
	type = character_event
	title = halfsade_captives.0007.t
	desc = halfsade_captives.0007.desc
	theme = diplomacy

	trigger = {
		any_courtier = { has_character_flag = halfsade_captives_leaving_pending }
	}

	left_portrait = {
		character = scope:halfsade_leaver
		animation = idle
	}

	immediate = {
		random_courtier = {
			limit = { has_character_flag = halfsade_captives_leaving_pending }
			save_scope_as = halfsade_leaver
			remove_character_flag = halfsade_captives_leaving_pending
		}
	}

	option = {
		name = halfsade_captives.0007.a
		scope:halfsade_leaver = {
			remove_character_modifier = halfsade_captives_freed_modifier
			move_to_pool = yes
		}
	}

	option = {
		name = halfsade_captives.0007.b
		trigger = { short_term_gold >= halfsade_captives_price_1 }
		show_as_unavailable = { always = yes }
		halfsade_captives_pay_effect = { COUNT = 1 }
		scope:halfsade_leaver = {
			remove_character_modifier = halfsade_captives_freed_modifier
		}
	}
}

halfsade_captives.0011 = {
	type = character_event
	hidden = yes

	trigger = {
		has_variable = halfsade_captives_keeper
		var:halfsade_captives_keeper = { is_alive = yes }
	}

	immediate = {
		var:halfsade_captives_keeper = { save_scope_as = halfsade_keeper }
		if = {
			limit = {
				NOT = { is_courtier_of = scope:halfsade_keeper }
				is_imprisoned = no
				is_ruler = no
				is_married = no
			}
			scope:halfsade_keeper = { add_courtier = root }
		}
		trigger_event = {
			id = halfsade_captives.0011
			days = 90
		}
	}
}
