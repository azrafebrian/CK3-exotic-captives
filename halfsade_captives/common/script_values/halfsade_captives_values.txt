# default price = duke rank (50 gold, 100 piety, 100 prestige)

# multiplier
halfsade_captives_cost_multiplier = {
	value = 0.4
	if = {
		limit = { highest_held_title_tier >= tier_county }
		add = 0.2
	}
	if = {
		limit = { highest_held_title_tier >= tier_duchy }
		add = 0.4
	}
	if = {
		limit = { highest_held_title_tier >= tier_kingdom }
		add = 1.0
	}
	if = {
		limit = { highest_held_title_tier >= tier_empire }
		add = 1.0
	}
	if = {
		# hegemony
		limit = { highest_held_title_tier >= 6 }
		add = 1.0
	}
}

halfsade_captives_gold_cost = {
	value = 50
	multiply = halfsade_captives_cost_multiplier
}

halfsade_captives_piety_cost = {
	value = 100
	multiply = halfsade_captives_cost_multiplier
}

halfsade_captives_prestige_cost = {
	value = 100
	multiply = halfsade_captives_cost_multiplier
}

halfsade_captives_gold_per_day = 2

halfsade_captives_captive_price = {
	value = var:halfsade_captives_travel_days
	if = {
		limit = { NOT = { has_variable = halfsade_captives_travel_days } }
		add = 20
	}
	multiply = halfsade_captives_gold_per_day
	multiply = halfsade_captives_cost_multiplier
}

halfsade_captives_price_1 = {
	value = halfsade_captives_captive_price
}

halfsade_captives_price_2 = {
	value = halfsade_captives_captive_price
	multiply = 2
}

halfsade_captives_price_3 = {
	value = halfsade_captives_captive_price
	multiply = 3
}
