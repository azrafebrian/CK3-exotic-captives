halfsade_captives_in_macro_europe_trigger = {
	trigger_if = {
		limit = { exists = capital_province }
		capital_province = {
			OR = {
				geographical_region = world_europe
			}
		}
	}
	trigger_else = {
		location = {
			OR = {
				geographical_region = world_europe
			}
		}
	}
}

halfsade_captives_option_available_europe_trigger = {
	trigger_if = {
		limit = { halfsade_captives_in_macro_europe_trigger = yes }
		always = no
	}
	trigger_else = {
		OR = {
			any_county_in_region = {
				region = world_europe_west_britannia
				halfsade_captives_county_valid_trigger = yes
			}

				region = world_europe_west_germania
