# check root home
halfsade_captives_in_macro_europe_trigger = {
	trigger_if = {
		limit = { exists = capital_province }
		capital_province = {
			OR = {
				geographical_region = world_europe
			}
		}
	}
	trigger_else = {
		location = {
			OR = {
				geographical_region = world_europe
			}
		}
	}
}
