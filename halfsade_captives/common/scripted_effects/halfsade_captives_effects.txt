halfsade_captives_create_captive_effect = {
	if = {
		limit = {
			scope:halfsade_home_ruler = { is_male = yes }
		}
		random_list = {
			85 = {
				halfsade_captives_roll_noble_effect = { SLOT = $SLOT$ GENDER = female }
			}
			15 = {
				halfsade_captives_roll_noble_effect = { SLOT = $SLOT$ GENDER = male }
			}
		}
	}
	else = {
		random_list = {
			85 = {
				halfsade_captives_roll_noble_effect = { SLOT = $SLOT$ GENDER = male }
			}
			15 = {
				halfsade_captives_roll_noble_effect = { SLOT = $SLOT$ GENDER = female }
			}
		}
	}
}

halfsade_captives_roll_noble_effect = {
	random_list = {
		15 = {
			halfsade_captives_spawn_captive_effect = { SLOT = $SLOT$ GENDER = $GENDER$ DYNASTY = generate }
		}
		85 = {
			halfsade_captives_spawn_captive_effect = { SLOT = $SLOT$ GENDER = $GENDER$ DYNASTY = none }
		}
	}
}

halfsade_captives_spawn_captive_effect = {
	create_character = {
		location = scope:halfsade_home_ruler.capital_province
		culture = scope:halfsade_far_prov_$SLOT$.culture
		faith = scope:halfsade_far_prov_$SLOT$.faith
		gender = $GENDER$
		age = { 16 30 }
		dynasty = $DYNASTY$
		random_traits = yes
		save_scope_as = halfsade_captive_$SLOT$
	}

	scope:halfsade_captive_$SLOT$ = {
		hidden_effect = {
			remove_trait = devoted
		}
	}

	scope:halfsade_captive_$SLOT$ = {
		halfsade_captives_slim_effect = yes
	}

	# 10% chance of a good inheritable trait
	scope:halfsade_captive_$SLOT$ = {
		random = {
			chance = 10
			random_list = {
				60 = { halfsade_captives_good_trait_effect = { LEVEL = 1 } }
				30 = { halfsade_captives_good_trait_effect = { LEVEL = 2 } }
				10 = { halfsade_captives_good_trait_effect = { LEVEL = 3 } }
			}
		}
	}
}

halfsade_captives_good_trait_effect = {
	random_list = {
		1 = { add_trait = beauty_good_$LEVEL$ }
		1 = { add_trait = intellect_good_$LEVEL$ }
		1 = { add_trait = physique_good_$LEVEL$ }
	}
}

# captives weight
halfsade_captives_slim_effect = {
	hidden_effect = {
		change_current_weight = -20
	}
}

halfsade_captives_welcome_effect = {
	add_courtier = scope:halfsade_captive_$SLOT$
	scope:halfsade_captive_$SLOT$ = {
		add_character_modifier = {
			modifier = halfsade_captives_far_from_home_modifier
			years = 5
		}
		add_character_modifier = {
			modifier = halfsade_captives_fine_lineage_modifier
		}
		# captives leash
		add_character_modifier = {
			modifier = halfsade_captives_belongs_modifier
		}
		set_variable = { name = halfsade_captives_keeper value = root }
		trigger_event = {
			id = halfsade_captives.0011
			days = 90
		}
	}
	add_character_modifier = {
		modifier = halfsade_captives_exotic_taste_modifier
		years = 2
	}
	add_character_modifier = {
		modifier = halfsade_captives_owner_modifier
		days = 100
	}
}

halfsade_captives_pay_effect = {
	remove_short_term_gold = halfsade_captives_price_$COUNT$
}

halfsade_captives_release_effect = {
	add_courtier = scope:halfsade_captive_$SLOT$
	scope:halfsade_captive_$SLOT$ = {
		add_character_modifier = {
			modifier = halfsade_captives_freed_modifier
			years = 1
		}
		add_character_modifier = {
			modifier = halfsade_captives_fine_lineage_modifier
		}

		trigger_event = {
			id = halfsade_captives.0006
			days = 360
		}
		trigger_event = {
			id = halfsade_captives.0012
			days = 30
		}
	}
	add_character_modifier = {
		modifier = halfsade_captives_exotic_taste_modifier
		years = 2
	}
}

halfsade_captives_send_home_effect = {
	scope:halfsade_captive_$SLOT$ = {
		death = {
			death_reason = death_vanished
		}
	}
}

halfsade_captives_pass_two_effect = {
	set_variable = { name = halfsade_captives_left_1 value = scope:halfsade_captive_$A$ }
	set_variable = { name = halfsade_captives_left_2 value = scope:halfsade_captive_$B$ }
	trigger_event = halfsade_captives.0004
}

halfsade_captives_pass_one_effect = {
	set_variable = { name = halfsade_captives_left_1 value = scope:halfsade_captive_$A$ }
	trigger_event = halfsade_captives.0005
}

halfsade_captives_cleanup_effect = {
	remove_variable = halfsade_captives_travel_days
	remove_variable = halfsade_captives_left_1
	remove_variable = halfsade_captives_left_2
	remove_variable = halfsade_captives_left_3
}

halfsade_captives_free_kept_effect = {
	remove_character_modifier = halfsade_captives_belongs_modifier
	remove_variable = halfsade_captives_keeper
	add_character_modifier = {
		modifier = halfsade_captives_freed_modifier
		years = 1
	}
	trigger_event = {
		id = halfsade_captives.0006
		days = 360
	}
	trigger_event = {
		id = halfsade_captives.0012
		days = 30
	}
}
