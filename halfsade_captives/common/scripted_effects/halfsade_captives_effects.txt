# root gender to captives gender
halfsade_captives_create_captive_effect = {
	if = {
		limit = {
			scope:halfsade_home_ruler = { is_male = yes }
		}
		random_list = {
			85 = {
				halfsade_captives_spawn_captive_effect = { SLOT = $SLOT$ GENDER = female }
			}
			15 = {
				halfsade_captives_spawn_captive_effect = { SLOT = $SLOT$ GENDER = male }
			}
		}
	}
	else = {
		random_list = {
			85 = {
				halfsade_captives_spawn_captive_effect = { SLOT = $SLOT$ GENDER = male }
			}
			15 = {
				halfsade_captives_spawn_captive_effect = { SLOT = $SLOT$ GENDER = female }
			}
		}
	}
}

halfsade_captives_spawn_captive_effect = {
	create_character = {
		location = scope:halfsade_home_ruler.capital_province
		culture = scope:halfsade_far_ruler_$SLOT$.culture
		faith = scope:halfsade_far_ruler_$SLOT$.faith
		gender = $GENDER$
		age = { 16 30 }
		dynasty = none
		random_traits = yes
		save_scope_as = halfsade_captive_$SLOT$
	}

	# 10% chance of a good inheritable trait
	scope:halfsade_captive_$SLOT$ = {
		random = {
			chance = 10
			random_list = {
				60 = { halfsade_captives_good_trait_effect = { LEVEL = 1 } }
				30 = { halfsade_captives_good_trait_effect = { LEVEL = 2 } }
				10 = { halfsade_captives_good_trait_effect = { LEVEL = 3 } }
			}
		}
	}
}

halfsade_captives_good_trait_effect = {
	random_list = {
		1 = { add_trait = beauty_good_$LEVEL$ }
		1 = { add_trait = intellect_good_$LEVEL$ }
		1 = { add_trait = physique_good_$LEVEL$ }
	}
}

halfsade_captives_welcome_effect = {
	add_courtier = scope:halfsade_captive_$SLOT$
	scope:halfsade_captive_$SLOT$ = {
		add_character_modifier = {
			modifier = halfsade_captives_far_from_home_modifier
			years = 5
		}
	}
	add_character_modifier = {
		modifier = halfsade_captives_exotic_taste_modifier
		years = 2
	}
}

halfsade_captives_send_home_effect = {
	scope:halfsade_captive_$SLOT$ = {
		death = {
			death_reason = death_vanished
		}
	}
}
