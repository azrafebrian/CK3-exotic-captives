halfsade_captives_try_macro_effect = {
	if = {
		limit = {
			NOT = { exists = scope:halfsade_far_ruler_$SLOT$ }
		}
		random_ruler = {
			limit = {
				halfsade_captives_culture_unused_trigger = yes
				halfsade_captives_in_macro_$MACRO$_trigger = yes
			}
			save_scope_as = halfsade_far_ruler_$SLOT$
			debug_log = "halfsade_captives: slot $SLOT$ picked from macro $MACRO$"
		}
	}
}

halfsade_captives_try_any_effect = {
